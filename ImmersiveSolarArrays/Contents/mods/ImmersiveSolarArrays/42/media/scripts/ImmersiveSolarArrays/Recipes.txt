/*
    Build 42 craft recipes.

    Changes from the build 41 originals and from the 42.1 beta branch:

      tags[...]        namespaced. Tag strings are matched literally after lowercasing,
                       so a bare "Screwdriver" matched nothing and every recipe that
                       asked for one was impossible to perform.
      OnCreate         points at a table this mod owns. Build 42 removed recipecode.lua
                       and the global Recipe table with it, so Recipe.OnCreate.* resolved
                       to nothing. The signature is (craftRecipeData, character) now.
      names            no spaces, and prefixed so they cannot collide with another mod's
                       recipe. The name is also the translation key, see Recipes.json.
      DIY battery      takes four wired car batteries again, as in build 41. The beta
                       changed it to a single raw car battery, which left its OnCreate
                       dividing by zero because a raw car battery carries no capacity.

    Make Solar Panel and Make Inverter are gated behind the enableExpandedRecipes
    sandbox option. Build 41 did that with recipe:setIsHidden, which no longer exists,
    so the gate is an OnTest that rejects every candidate item while the option is off.
*/
module ISA

    craftRecipe ISA_WireCarBattery
    {
        timedAction = MakingElectrical,
        time = 80,
        category = Electrical,
        NeedToBeLearn = true,
        AutoLearnAny = Electricity:4,
        SkillRequired = Electricity:3,
        xpAward = Electricity:1,
        OnCreate = ISARecipes.OnCreate.wireCarBattery,
        Tags = AnySurfaceCraft,
        inputs
        {
            item 1 tags[base:screwdriver] mode:keep flags[Prop1;MayDegradeLight;NoBrokenItems],
            item 1 [Base.ElectricWire],
            item 1 tags[base:carbattery] mode:destroy flags[NoBrokenItems],
        }
        outputs
        {
            item 1 ISA.WiredCarBattery,
        }
